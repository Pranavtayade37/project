# ================================
# SALES DATA ANALYSIS PROJECT
# ================================

# Step 1: Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Display settings
plt.figure(figsize=(8,5))

# ================================
# Step 2: Create Sample Dataset
# ================================
data = {
"Order Date": ["2023-01-10","2023-02-15","2023-03-20","2023-04-10","2023-05-05",
"2023-06-18","2023-07-22","2023-08-14","2023-09-30","2023-10-12"],
"Category": ["Technology","Furniture","Office Supplies","Technology","Furniture",
"Technology","Office Supplies","Furniture","Technology","Office Supplies"],
"Region": ["East","West","South","North","East","West","South","North","East","West"],
"Sales": [200,300,150,400,250,500,180,350,600,220]
}

df = pd.DataFrame(data)

# Save dataset (optional)
df.to_csv("sales_data.csv", index=False)

print("Dataset Created:\n")
print(df.head())

# ================================
# Step 3: Understand Dataset
# ================================
print("\nDataset Info:\n")
print(df.info())

print("\nStatistical Summary:\n")
print(df.describe())

# ================================
# Step 4: Data Cleaning
# ================================
print("\nChecking Null Values:\n")
print(df.isnull().sum())

# Drop nulls (if any)
df = df.dropna()

# Remove duplicates
df = df.drop_duplicates()

# ================================
# Step 5: Data Preprocessing
# ================================
df['Order Date'] = pd.to_datetime(df['Order Date'])

df['Year'] = df['Order Date'].dt.year
df['Month'] = df['Order Date'].dt.month

print("\nAfter Preprocessing:\n")
print(df.head())

# ================================
# Step 6: Data Analysis
# ================================

# Total Sales
total_sales = df['Sales'].sum()
print("\nTotal Sales:", total_sales)

# Sales by Category
category_sales = df.groupby('Category')['Sales'].sum()
print("\nSales by Category:\n", category_sales)
print("\nProject Completed Successfully!")
